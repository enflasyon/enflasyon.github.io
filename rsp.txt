class Surface(QWidget):
    """Keep the original 1840 x 1000 coordinate map; scale as one layout."""
    def __init__(self):
        super().__init__()
        self.setObjectName('surface')
        self.setAttribute(Qt.WA_StyledBackground, True)
        self.setMinimumSize(1400, 850)
        self.placements = []

    def put(self, widget, x, y, width, height):
        widget.setParent(self)
        self.placements.append((widget, x, y, width, height))
        self.reflow()
        return widget

    def reflow(self):
        sx, sy = self.width()/1840, self.height()/1000
        for w, x, y, width, height in self.placements:
            w.setGeometry(round(x*sx), round(y*sy), round(width*sx), round(height*sy))

    def resizeEvent(self, event):
        super().resizeEvent(event)
        self.reflow()

    def label(self, text, rect, heading=False, right=False):
        w = QLabel(text)
        w.setProperty('heading', heading)
        w.setAlignment((Qt.AlignRight if right else Qt.AlignLeft) | Qt.AlignVCenter)
        w.setToolTip(text)
        return self.surface.put(w, *rect)

    def button(self, text, rect, action, icon=None, primary=False):
        w = QPushButton(text)

            w.setIconSize(QSize(17, 17))
        w.setCursor(Qt.PointingHandCursor)
        w.clicked.connect(lambda checked=False: self.invoke(action))
        self.surface.put(w, *rect)
        return w

    def invoke(self, action):
        """Qt button boundary: report errors instead of terminating the GUI."""
        try:
            action()
        except (Exception, SystemExit) as exc:
            traceback.print_exc()
            QMessageBox.critical(self, 'RSP Plotter', str(exc))
            self.statusBar().showMessage('Operation failed — see error details')

    def bt0_command(self):
        path, _ = QFileDialog.getOpenFileName(self, 'Import RSP / RPC file', '',
                                            'RPC data (*.rsp *.rpc *.tim);;All files (*)')
        if not path:
            return
        self.statusBar().showMessage('Reading file…')
        QApplication.setOverrideCursor(Qt.WaitCursor)
        try:
            reader = rpc_reader.ReadRPC(path, extra_headers=[
                ('DATA_TYPE', 'SHORT_INTEGER'), ('INT_FULL_SCALE', '32768')])
            reader.import_rpc_data_from_file()
            channels = reader.get_channels()
            data, time = reader.get_data(), reader.get_time()
            names, units = [], []
            for key, value in channels.items():
                names.append(value['Description'])
                units.append(value['Units'])
        finally:
            QApplication.restoreOverrideCursor()
        self.head, self.tail = os.path.split(path)
        self.data, self.Z, self.D, self.U = data, time, names, units
        self.lab0.setText(self.tail)
        self.lab0.setToolTip(path)
        self.lb1.clear()
        self.lb2.clear()
        self.lb1.addItems(names + ['Time'])
        self.lb2.addItems(names + ['Time'])
        self.lb1select, self.lb2select = [], []
        self.plot_ready = False
        self.bt4_command()
        self.statusBar().showMessage(f'{self.tail}  •  {len(names)} channels loaded')

    def bt1_command(self):
        selected = self.lb1.selectedItems()
        if not selected:
            raise ValueError('Select one X channel first.')
        self.lb1select = [w.text() for w in selected]
        self.lab3.setText('Selected X Data\n\n' + self.lb1select[0])
        self.plot_ready = False

    def bt2_command(self):
        selected = sorted(self.lb2.selectedItems(), key=self.lb2.row)
        if not selected:
            raise ValueError('Select at least one Y channel first.')
        self.lb2select = [w.text() for w in selected]
        self.units = [self.lb2.row(w) for w in selected]
        self.lab4.setText('Selected Y Data\n\n' + '\n'.join(self.lb2select))
        self.lab4.setToolTip('\n'.join(self.lb2select))
        self.plot_ready = False

    def bt3_command(self):
        if self.data is None or not self.lb1select or not self.lb2select:
            raise ValueError('Import a file, then use Get X Values and Get Y Values.')
        self.plot_ready = False
        self.ax.clear()
        self._style_axes()
        self.Table_List, self.aydata = [], []
        self.X_Data_Name = self.lb1select[0]
        self.axdata = self.Z[0] if self.X_Data_Name == 'Time' else self.data[:, self.D.index(self.X_Data_Name)]
        for name in self.lb2select:
            if name == 'Time':
                values, unit = self.Z[0], 'Sec'
            else:
                idx = self.D.index(name)
                values, unit = self.data[:, idx], self.U[idx]
            self.aydata.append(values)
            self.Table_List.append([name, min(values), max(values), '['+unit+']'])
            self.ax.plot(self.axdata, values, label=name+'['+unit+']', linewidth=.5 if name != 'Time' else 1.5)
        self.ax.legend(loc='upper left', frameon=False, fontsize=9)
        xunit = 's' if self.X_Data_Name == 'Time' else self.U[self.D.index(self.X_Data_Name)]
        self.ax.set_xlabel(self.X_Data_Name+' ['+xunit+']', color='#24374f')
        yunits = {r[3] for r in self.Table_List}
        self.ax.set_ylabel('Y Data '+(next(iter(yunits)) if len(yunits)==1 else ''), color='#24374f')
        self.canvas1.draw_idle()
        self.tree.setRowCount(len(self.Table_List))
        for row, values in enumerate(self.Table_List):
            for col, value in enumerate(values):
                self.tree.setItem(row, col, QTableWidgetItem(f'{value:.6g}' if isinstance(value, (float, np.floating)) else str(value)))
        self.lab3.setText('Selected X Data\n\n'+self.X_Data_Name)
        self.lab4.setText('Selected Y Data\n\n'+'\n'.join(self.lb2select))
        self.plot_ready = True
        self.statusBar().showMessage(f'{len(self.lb2select)} curves  •  {len(self.axdata):,} samples  •  raw data')

    def bt4_command(self):
        self.lab3.setText('Selected X Data')
        self.lab4.setText('Selected Y Data')
        self.tree.setRowCount(0)
        self._empty_plot()
        self.plot_ready = False

    def _require_plot(self):
        if not self.plot_ready:
            raise ValueError('Use Import Plot before exporting or calculating.')

    def bt5_command(self):  # Original export averaging retained
        self._require_plot()
        dict1 = {self.X_Data_Name:self.axdata}
        for i in range(len(self.lb2select)):
            dict1.update({str(self.Table_List[i][0]+self.Table_List[i][3]): self.aydata[i]})
        df = pd.DataFrame(dict1)
        Dpsec=len(df.index)/max(self.axdata)
        Freq=1  # New sampling frequency per second with mean value
        NS=Dpsec/Freq  # Define new division
        s = (df.index.to_series() / int(NS)).astype(int)  # Create Label group
        df = df.groupby(s).mean().set_index(s.index[0::int(NS)])  # Get New Sampling by using mean value
        df.to_csv(self.tail[:-4]+'.csv', encoding='utf-8', index=False)
        fig = px.line(df, x = df.columns[0], y = df.columns[1:len(df.columns)], title=self.tail)
        fig.update_traces(mode="lines")
        fig.update_traces(line=dict(width=2))
        fig.update_xaxes(showspikes=True,spikecolor='grey',spikesnap='cursor')
        fig.update_layout(spikedistance=100, hoverdistance=100)
        fig.update_layout(hovermode='x')
        fig.write_html(self.tail[:-4]+'.html')
        fig.show()

    def bt6_command(self):  # Original calculation expressions retained
        self._require_plot()
        if len(self.lb2select) != 3:
            raise ValueError("Select exactly three rosette channels.")
        names = [r[0] for r in self.Table_List]
        if not (names[0][:-1] == names[1][:-1] == names[2][:-1] and
                [n[-1:] for n in names] == ["1", "2", "3"] and
                len({r[3] for r in self.Table_List}) == 1):
            raise ValueError("Select matching rosette channels ending in 1, 2, 3 with identical units.")
        dict1 = {self.X_Data_Name:self.axdata}
        if len(self.lb2select)>3:
            return

        if self.Table_List[0][3]==self.Table_List[1][3]==self.Table_List[2][3]:
            if self.Table_List[0][0][:-1]==self.Table_List[1][0][:-1]==self.Table_List[2][0][:-1]:
                if self.Table_List[0][0][-1]=="1" and self.Table_List[1][0][-1]=="2" and self.Table_List[2][0][-1]=="3":
                    self.EX=[]
                    self.EY=[]
                    self.EXY=[]
                    self.SX=[]
                    self.SY=[]
                    self.SXY=[]
                    self.S1=[]
                    self.S3=[]
                    self.Seqv=[]

                    Ang1=float(self.Ent1.text())
                    Ang2=float(self.Ent2.text())
                    Ang3=float(self.Ent3.text())
                    Ang4=float(self.Ent4.text())
                    E=float(self.Ent5.text())
                    v=float(self.Ent6.text())

                    Ang1=math.radians(Ang1)
                    Ang2=math.radians(Ang2)
                    Ang3=math.radians(Ang3)
                    Ang4=math.radians(Ang4)

                    A=np.array([[math.cos(Ang1)**2, math.sin(Ang1)**2, math.cos(Ang1)*math.sin(Ang1)],
                                [math.cos(Ang2)**2, math.sin(Ang2)**2, math.cos(Ang2)*math.sin(Ang2)],
                                [math.cos(Ang3)**2, math.sin(Ang3)**2, math.cos(Ang3)*math.sin(Ang3)]])

                    B=np.array([self.aydata[0],self.aydata[1],self.aydata[2]])
                    X = np.linalg.solve(A,B)

                    self.EX=X[0]
                    self.EY=X[1]
                    self.EXY=X[2]

                    #Coordinate Correction
                    self.EX=(self.EX+self.EY)/2+((self.EX-self.EY)/2)*math.cos(2*Ang4)+self.EXY*math.sin(2*Ang4)
                    self.EY=(self.EX+self.EY)/2-((self.EX-self.EY)/2)*math.cos(2*Ang4)-self.EXY*math.sin(2*Ang4)
                    self.EXY=-((self.EX-self.EY)/2)*math.sin(2*Ang4)+self.EXY*math.cos(2*Ang4)
                    self.SX=((self.EX*(10**-6)+self.EY*(10**-6)*v)*E)/(1-v**2)  # Component X Stress
                    self.SY=((self.EY*(10**-6)+self.EX*(10**-6)*v)*E)/(1-v**2)  # Component Y Stress
                    self.SXY=(self.EXY*(10**-6))*E/(2+2*v)  # Shear Stress
                    self.S1=((self.SX+self.SY)/2)+(((self.SX-self.SY)/2)**2+self.SXY**2)**0.5  # Max Principle Stress
                    self.S3=((self.SX+self.SY)/2)-(((self.SX-self.SY)/2)**2+self.SXY**2)**0.5  # Min Principle Stress
                    self.Seqv=(self.S1**2-self.S1*self.S3+self.S3**2)**0.5  # Equivalent Stress

                    for i in range(len(self.lb2select)):
                        dict1.update({str(self.Table_List[i][0]+self.Table_List[i][3]): self.aydata[i]})
                    dict1.update({str(self.Table_List[0][0][:-1])+'εX': self.EX})
                    dict1.update({str(self.Table_List[0][0][:-1])+'εY': self.EY})
                    dict1.update({str(self.Table_List[0][0][:-1])+'εXY': self.EXY})
                    dict1.update({str(self.Table_List[0][0][:-1])+'SX': self.SX})
                    dict1.update({str(self.Table_List[0][0][:-1])+'SY': self.SY})
                    dict1.update({str(self.Table_List[0][0][:-1])+'SXY': self.SXY})
                    dict1.update({str(self.Table_List[0][0][:-1])+'S1': self.S1})
                    dict1.update({str(self.Table_List[0][0][:-1])+'S3': self.S3})
                    dict1.update({str(self.Table_List[0][0][:-1])+'SEqv': self.Seqv})
                    df = pd.DataFrame(dict1)
                    Dpsec=len(df.index)/max(self.axdata)
                    Freq=1  # New sampling frequency per second with mean value
                    NS=Dpsec/Freq  # Define new division
                    s = (df.index.to_series() / int(NS)).astype(int)
                    df = df.groupby(s).mean().set_index(s.index[0::int(NS)])
                    fig = px.line(df, x = df.columns[0], y = df.columns[1:len(df.columns)], title=self.tail)
                    fig.update_traces(mode="lines")
                    fig.update_traces(line=dict(width=2))
                    fig.update_xaxes(showspikes=True,spikecolor='grey',spikesnap='cursor')
                    fig.update_layout(spikedistance=100, hoverdistance=100)
                    fig.update_layout(hovermode='x')
                    fig.write_html(str(self.Table_List[0][0][:-1])+'Calculated'+'.html')
                    fig.show()
